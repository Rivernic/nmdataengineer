# one row of the enrichment pipeline
def enrich_lead(company: str, role: str) -> dict:
    match = apollo.find_contact(company, role)
    email = verify.check(match.email)

    return {
        "company": company,
        "email": email.address,
        "status": "VERIFIED" if email.deliverable else "SKIPPED",
        "linkedin": match.linkedin_url,
    }
